Program 13 - Find the Greatest of Three Numbers

Problem Statement

Enter three numbers from users and tell which is greater.

Logic

Compare the three numbers and find the greatest number. 

#include <stdio.h>

void main()
{
	int num1,num2,num3;

	printf("Enter 1st number: ");
	scanf("%d",&num1);
	printf("Enter 2nd number: ");
	scanf("%d",&num2);
	printf("Enter 3rd number: ");
	scanf("%d",&num3);
	
	if(num1 >= num2 && num1 >= num3)
	  printf("Greatest = %d",num1);
	else if(num2 >= num1 && num2 >= num3)
	  printf("Greatest = %d",num2);
	else
	  printf("Greatest = %d",num3);
}
