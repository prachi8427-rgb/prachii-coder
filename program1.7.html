Program 07 - Find Power of a Number

Problem Statement

Make function for find out power of number, enter the number and power from user.

Source Code

#include <stdio.h>

int power(int a, int b)
{
	int i, result = 1;
	for(i = 1; i <= b; i++)
	{
		result = result * a;
	}

	return result;
}

void main()
{
	int base, exp, res;

	printf("Enter base: ");
	scanf("%d",&base);
	printf("Enter exponent: ");
	scanf("%d",&exp);
	
	res = power(base,exp);

	printf("\n%d raised to the power %d = %d",base ,exp ,res);

}
