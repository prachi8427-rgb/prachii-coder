Program 02 - Print Only Names of Employees

Problem Statement

Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print only name of the employee.

Source Code

#include <stdio.h>

struct emp
{
	int id;
	char name[20];
	char designation[30];
};

void main()
{
	struct emp e[50];
	int i;

	for(i = 0; i < 50; i++)
	{
		printf("\nEnter details of employee %d\n", i + 1);

		printf("Enter Employee Id: ");
		scanf("%d",&e[i].id);

		printf("Enter Name: ");
		scanf("%s",e[i].name);

		printf("Enter Designation: ");
		scanf("%s",e[i].designation);
	}

	printf("\nNames of Employees:\n");
	printf("---------------------\n");
	for(i = 0; i < 50; i++)
	{
	  printf("%s\n",e[i].name);
	}
}
