Program 20 - Check Whether a Date is Vaild

Problem Statement

Enter date, month, year and check it is valid date or not.

Logic

Check the month, the number of days in that month, and whether february has 28 or 29 days. 

#include <stdio.h>

void main()
{
	int d,m,y,maxdays;

	printf("Enter days: ");
	scanf("%d",&d);
	printf("Enter Month: ");
	scanf("%d",&m);
	printf("Enter year: ");
	scanf("%d",&y);

	if(m < 1 || m > 12 || d < 1 || y < 1)
		printf("Invalid date");
	else
	{
		if(m == 4 || m == 6 || m == 9 || m == 11)
		  maxdays = 30;
		else if(m == 2)
		{
			if(y % 400 == 0 || (y % 4 == 0 && y % 100))
			  maxdays = 29;
			else 
			  maxdays = 28;
		}  

		else 
		  maxdays = 31;

		if(d <= maxdays)
		  printf("Valid date");
		else 
		  printf("Invalid date");
	}
}
