Program 12 - Menu-Driven Calculator

Problem Statement

Enter a number from the user if the user enter 1 then you will perform addition operation by entering 2 number from user add it and display answer. if user enter 2 then subtract or if user enter 3 then multiply or if user enter 4 then divide or if user enter 5 then modulo or if any other number then print invalid .if user want to quit the program then user will enter Q.

Logic

Use a do-while loop to display the menu repeatedly and perform the selected operation using switch-case.

Source Code

#include <stdio.h>
#include <conio.h>

void main()
{
	int a, b;
	char ch;

	clrscr();

	do{
		printf("\n1. Addition");
		printf("\n2. Subtraction");
		printf("\n3. Multiplication");
		printf("\n4. Division");
		printf("\n5. Modulo");
		printf("\nEnter your choice or Q to quit: ");
		ch=getche();

		if(ch == 'Q' || ch == 'q')
		break;

		if(ch >= '1' && ch <= '5')
		{
			printf("\nEnter 1st number: ");
			scanf("%d",&a);
			printf("Enter 2nd number: ");
			scanf("%d",&b);

			switch(ch)
			{
				case '1':
				  printf("Addition = %d", a + b);
				  break;

				case '2':
				  printf("Subtraction = %d", a - b);
				  break;

				case '3':
				  printf("Multiplication = %d", a * b);
				  break;

				case '4':
				if(b != 0)
				  printf("Division = %d", a / b);
				else
				  printf("Division by Zero is not possible.");
				  break;

				case '5':
				  if(b != 0)
				  printf("Modulo = %d", a % b);
				else
				  printf("Modulo by Zero is not possible.");
				  break;
			}
		}

		else
		printf("\nInvalid Choice!");
	} while(1);
	
	printf("\nProgram Quit.");
}
