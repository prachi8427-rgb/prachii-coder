Program 06 - Swap Numbers Using Call by function

Problem Statement

Write a program to swap numbers using Call by function.

Source Code

#include <stdio.h>

void swap(int a, int b)
{
	int temp;

	temp = a;
	a = b;
	b = temp;

	printf("\nAfter swapping inside function:\n");
	printf("First number = %d\n",a);
	printf("Second number = %d\n",b);
}

void main()
{
	int x, y;

	printf("Enter first number: ");
	scanf("%d",&x);
	printf("Enter second number: ");
	scanf("%d",&y);

	printf("\nBefore swapping:\n");
	printf("First number = %d\n",x);
	printf("Second number = %d\n",y);

	swap(x,y);
}
