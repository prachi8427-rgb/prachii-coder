Program 04 - Factorial, Reverse, Palindrome, Prime, Armstrong and Fibonacci

Problem Statement

Making following program using local and global variable ? Factorial , Reverse , Palindrome , Prime numbers, Armstrong , Fibonacci.

Source Code

#include <stdio.h>
int num;

void input()
{
	printf("Enter a number: ");
	scanf("%d",&num);
}

void fact()
{
	int i, fact = 1;
	for(i = 1; i <= num; i++)
	  fact = fact * i;

	printf("Factorial = %d\n",fact);
}

void reverse()
{
	int n, rem, rev = 0;
	n = num;

	while(n > 0)
	{
		rem = n % 10;
		rev = rev * 10 + rem;
		n = n / 10;
	}

	printf("Reverse = %d\n", rev);
}

void palindrome()
{
	int n, rem, rev = 0;
	n = num;

	while(n > 0)
	{
		rem = n % 10;
		rev = rev * 10 + rem;
		n = n / 10;
	}

	if(num == rev)
	  printf("Palindrome Number\n");
	else
	  printf("Not a Palindrome Number\n");
}

void prime()
{
	int i, count = 0;

	for(i = 1; i <= num; i++)
	{
		if(num % i == 0)
		count++;
	}

	if(count == 2)
	  printf("Prime Number\n");
	else
	  printf("Not a Prime Number\n");
}

void armstrong()
{
	int n, rem, sum = 0;
	n = num;

	while(n > 0)
	{
		rem = n % 10;
		sum = sum + rem * rem * rem;
		n = n /10;
	}

	if(num == sum)
	  printf("Armstrong number\n");
	else
	  printf("Not an Armstrong number\n");
}

void fibonacci()
{
	int a = 0, b = 1, c, i;

	printf("Fibonacci Series: ");
	for(i = 1; i <= num; i++)
	{
		printf("%d ",a);
		c = a + b;
		a = b;
		b = c;
	}

	printf("\n");
}

void main()
{
	input();
	fact();

	input();
	reverse();

	input();
	palindrome();

	input();
	prime();

	input();
	armstrong();

	input();
	fibonacci();
}
