Program 04 - Student Details in Ascending Order of Roll Number

Problem Statement

Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number.

Source Code

#include <stdio.h>

struct stu
{
	int roll;
	char name[20];
	float fee;
	char dob[15];
};

void main()
{
	struct stu s[50], temp;
	int i, j, n;

	printf("Enter number of student(50 max): ");
	scanf("%d",&n);

	for(i = 0; i < n; i++)
	{
		printf("\nEnter details of student %d\n", i + 1);

		printf("Enter Roll Number: ");
		scanf("%d",&s[i].roll);
		fflush(stdin);

		printf("Enter Name: ");
		gets(s[i].name);

		printf("Enter Fee: ");
		scanf("%f",&s[i].fee);
		fflush(stdin);

		printf("Enter DOB (DD-MM-YYYY): ");
		gets(s[i].dob);
	}

	for(i = 0; i < n - 1; i++)
	{
		for(j = i + 1; j < n; j++)
		{
			if(s[i].roll > s[j].roll)
			{
				temp = s[i];
				s[i] = s[j];
				s[j] = temp;
			}
		}
	}



	printf("\n\nSTUDENT DETAILS\n");
	printf("---------------------------------------------------\n");
	printf("Roll\tNAME\t\tFEE\t\tDOB\n");
	printf("---------------------------------------------------\n");

	for(i = 0; i < n; i++)
	{
	  printf("%d\t%-15s\t%.2f\t%s\n", s[i].roll, s[i].name, s[i].fee, s[i].dob);
	}
}
