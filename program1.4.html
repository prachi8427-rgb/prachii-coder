Program 04 - Subtract Two Matrices

Problem Statement

Write a program to enter 2 matrix from user and subtract them.

Logic

Subtract the corresponding elements of both matrices and store the result in a third matrix.

Source Code

#include <stdio.h>

void main()
{
	int num[2][2], b[2][2], sub[2][2], i, j;

	printf("Enter 4 values in first matrix: \n");
	
	for(i = 0; i <= 1; i++)
	{
		for(j = 0; j<=1; j++)
		{
			scanf("%d",&num[i][j]);
		}
	}
	printf("Enter 4 values in second matrix: \n");
	
	for(i = 0; i <= 1; i++)
	{
		for(j = 0; j<=1; j++)
		{
			scanf("%d",&b[i][j]);
		}
	}


	for(i = 0; i <= 1; i++)
	{
		for(j = 0; j<=1; j++)
		{
			sub[i][j] = num[i][j] - b[i][j];
		}
	}


	printf("\nSubtraction of Matrices : \n");
	for(i = 0; i <= 1; i++)
	{
		for(j = 0; j<=1; j++)
		{
			printf("%d ",&sub[i][j]);
		}
		printf("\n");
	}	
}
